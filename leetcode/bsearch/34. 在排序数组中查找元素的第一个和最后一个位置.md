## 

#  34. 在排序数组中查找元素的第一个和最后一个位置

[力扣题目链接(opens new window)](https://leetcode.cn/problems/find-first-and-last-position-of-element-in-sorted-array/)

## 题解 -- 直接套用acwing的两个不同边界的二分查找。当然要注意判断边界情况

```
class Solution {
public:
    vector<int> searchRange(vector<int>& nums, int target) {
        if (nums.size() == 0) return {-1, -1};

        int l = 0;
        int r = nums.size() - 1;
        while (l < r)
        {
            int mid = (l + r) >> 1;
            if (nums[mid] >= target) r = mid;
            else l = mid + 1;
        }

        int result_left = l; // 获得左边界
        if (nums[l] != target) return {-1, -1};

        l = 0;
        r = nums.size() - 1;
        while (l < r)
        {
            int mid = (l + r + 1) >> 1;
            if (nums[mid] <= target) l = mid;
            else r = mid - 1;
        }

        int result_right = l;
        return {result_left, result_right};
    }
};
```
