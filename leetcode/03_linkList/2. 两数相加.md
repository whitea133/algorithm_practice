## 2. 两数相加

题目链接：https://leetcode.cn/problems/add-two-numbers/description/?envType=study-plan-v2&envId=top-100-liked

## 题解

c++:

```
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* addTwoNumbers(ListNode* l1, ListNode* l2) {
        ListNode *head = nullptr, *tail = nullptr; // head结果链表头结点；tail结果链表尾结点，用来尾插
        int carry = 0; // 保存进位，相加最多进1

        // 只要l1没走完 或者 l2没走完，循环继续
        while (l1 || l2) 
        {
            // 如果链表不为空取当前val，为空就补0
            int n1 = l1 ? l1->val: 0;
            int n2 = l2 ? l2->val: 0;

            int sum = n1 + n2 + carry; // 当前位 = l1数字 + l2数字 + 上一轮的进位

            if (!head) 
            {
                // 结果链表为空，插入第一个结点，head和tail同时指向它
                head = tail = new ListNode(sum % 10);
            } 
            else 
            {
                // 尾插法：在tail后面新建结点，再把tail向后移动
                tail->next = new ListNode(sum % 10);
                tail = tail->next;
            }

            carry = sum / 10; // 计算新的进位（sum>=10则carry=1，否则0）

            // l1向后走，不为空才移动，防止空指针访问
            if (l1) {
                l1 = l1->next;
            }
            // l2向后走
            if (l2) {
                l2 = l2->next;
            }
        }

        // 循环结束，如果还有剩余进位（例如 999+999，最后会多出进位1），需要再补一个结点
        if (carry > 0) {
            tail->next = new ListNode(carry);
        }

        return head; // 返回结果链表的头节点
    }
};

```
