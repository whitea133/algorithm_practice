代码如下：

```
class Solution {
private:
    // 方向数组：右、下、左、上，四个移动方向 {行偏移，列偏移}
    static constexpr int directions[4][2] = {{0, 1}, {1, 0}, {0, -1}, {-1, 0}}; // 这里改成const int也行
public:
    vector<int> spiralOrder(vector<vector<int>>& matrix) {
        // 边界判断：空矩阵 或者 矩阵第一行为空，直接返回空数组
        if (matrix.size() == 0 || matrix[0].size() == 0) {
            return {};
        }

        int rows = matrix.size();         // 矩阵行数
        int columns = matrix[0].size();   // 矩阵列数
        // visited标记矩阵：记录该位置是否已经被访问过
        vector<vector<bool>> visited(rows, vector<bool>(columns));
        int total = rows * columns;       // 矩阵总元素个数，循环一共执行total次
        vector<int> order(total);         // 保存螺旋遍历输出的结果

        int row = 0, column = 0;          // 当前所在的坐标位置，起点 (0,0)
        int directionIndex = 0;           // 当前使用哪个方向，0右，1下，2左，3上

        // 一共访问 total 个元素，循环total次
        for (int i = 0; i < total; i++) {
            order[i] = matrix[row][column];      // 将当前位置元素存入结果数组
            visited[row][column] = true;         // 标记当前坐标已经访问过

            // 按照当前方向，计算下一个候选坐标
            int nextRow = row + directions[directionIndex][0];
            int nextColumn = column + directions[directionIndex][1];

            // 判断下一个坐标是否越界，或者已经访问过
            // 如果越界 / 已经走过：需要顺时针切换下一个方向
            if (nextRow < 0 || nextRow >= rows || nextColumn < 0 || nextColumn >= columns || visited[nextRow][nextColumn]) {
                directionIndex = (directionIndex + 1) % 4; // 模4：0→1→2→3→0循环切换方向
            }

            // 更新当前坐标：使用更新后的方向，移动到下一个真实位置
            row += directions[directionIndex][0];
            column += directions[directionIndex][1];
        }
        return order;
    }
};

我的写法：

```
class Solution {
private:
    const int dirs[4][2] = {{0, 1}, {1, 0}, {0, -1}, {-1, 0}};


public:
    vector<int> spiralOrder(vector<vector<int>>& matrix) {
        if (matrix.size() == 0 || matrix[0].size() == 0) return {};

        int rows = matrix.size();
        int cols = matrix[0].size();
        vector<vector<bool>> visited(rows, vector<bool> (cols, false));
        int total = rows * cols;
        vector<int> order(total);

        int x = 0, y = 0;
        int dirs_index = 0;

        for (int i = 0; i < total; i++)
        {
            order[i] = matrix[x][y];
            visited[x][y] = true;

            int nextx = x + dirs[dirs_index][0];
            int nexty = y + dirs[dirs_index][1];

            if (nextx < 0 || nextx >= rows || nexty < 0 || nexty >= cols || visited[nextx][nexty]) dirs_index = (dirs_index + 1) % 4;

            x += dirs[dirs_index][0];
            y += dirs[dirs_index][1];
        }

        return order;
    }
};
