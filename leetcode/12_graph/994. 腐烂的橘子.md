# 994. 腐烂的橘子

题目：https://leetcode.cn/problems/rotting-oranges/description/?envType=study-plan-v2&envId=top-100-liked

## 题解 -- 借助BFS搜索获得最小值

代码如下：

```
class Solution {
public:
    int orangesRotting(vector<vector<int>>& grid) {
        int min = 0, fresh = 0;
        queue<pair<int,int>> q; // 队列存腐烂的橘子的坐标
        for (int i = 0; i  < grid.size(); i++)
        {
            for (int j = 0; j < grid[0].size(); j++)
            {
                if (grid[i][j] == 1) fresh++;
                else if (grid[i][j] == 2) q.push({i, j});
            }
        }

        int dir[4][2] = {0, 1, 1, 0, -1, 0, 0, -1};
        while (!q.empty())
        {
            int n = q.size();
            bool rotten = false;
            for (int i = 0; i < n; i++)
            {
                auto x = q.front();
                q.pop();
                for (int t = 0; t < 4; t++)
                {
                    int nextx = x.first + dir[t][0];
                    int nexty = x.second + dir[t][1];
                    if (nextx >= 0 && nextx < grid.size() && nexty >= 0 && nexty < grid[0].size() && grid[nextx][nexty] == 1)
                    {
                        grid[nextx][nexty] = 2;
                        q.push({nextx, nexty});
                        fresh--;
                        rotten = true;
                    }
                }
            }
            if (rotten) min++;
        }

       return fresh ? -1 : min;
    }
};
```
